#include "UpdateStatus.h"
#include "config.h"
#include <Arduino.h>
#include <WiFi.h>
#include <PubSubClient.h> // nécessaire pour PubSubClient

// Constructeur avec référence MQTT
UpdateStatus::UpdateStatus(PubSubClient &mqttClient) : mqtt(mqttClient) {}

// Connexion au broker MQTT
void UpdateStatus::connectMQTT() {
    mqtt.setServer(MQTT_HOST, MQTT_PORT);
    mqtt.setBufferSize(4096);
    mqtt.setKeepAlive(30);

    while (!mqtt.connected()) {
        Serial.print("🔗 Connexion MQTT...");
        if (mqtt.connect("ESP32ClientStone", MQTT_USER, MQTT_PASSWD)) {
            Serial.println("  MQTT connecté");
            mqtt.subscribe(TOPIC_SCANS_RESPONSE);
            mqtt.subscribe(TOPIC_SCANS_DYNAMIC_RESPONSE);

            // demande la liste au démarrage
            mqtt.publish(TOPIC_SCANS, "Update status");
        } else {
            Serial.printf(" rc=%d → retry 5s\n", mqtt.state());
            delay(5000);
        }
    }
}

// Envoie le statut au broker
void UpdateStatus::sendStatus(bool isEmpty) {
    // Construction JSON simple
    String payload = "{\"name_soap_dispenser\":\"";
    payload += DISPENSER_NAME;
    payload += "\",\"isEmpty\":";
    payload += isEmpty ? "true" : "false";
    payload += "}";

    mqtt.publish(TOPIC_SCANS, payload.c_str());
    Serial.println("📤 Statut envoyé : " + payload);
}

// Exemple de topic : "/scan/12"
const topicParts = msg.topic.split("/");
// topicParts = ["", "scan", "12"]

const id = topicParts[2]; // "12"

if (!id) {
    node.warn("ID non trouvé dans le topic !");
    return null;
}

// Vérifier que c'est bien un nombre
if (isNaN(id)) {
    node.warn("L'ID n'est pas un nombre !");
    return null;
}

// 🔍 Affichage de l'ID extrait
node.warn("ID extrait du topic MQTT : " + id);

// Construction de l'URL avec l'ID numérique
const url = "http://localhost:3001/api/dispenser/" + id;

msg.url = url;

// 🔧 Log de l'URL finale
node.warn("URL construite : " + url);

return msg;

#include <Arduino.h>
#include <WiFi.h>
#include <PubSubClient.h>
#include "config.h"
#include "Login.h"
#include "WaterSensor.h"
#include "UpdateStatus.h"

// ===== Objets =====
WaterSensor monCapteur(23);
Login login;

WiFiClient espClient;
PubSubClient mqtt(espClient);
UpdateStatus updater(mqtt);

// ===== Variable pour suivre l'état précédent =====
bool lastIsEmpty = true; // initial, supposons que le distributeur est vide

void setup() {
    Serial.begin(115200);

    // Connexion WiFi
    WiFi.begin(WIFI_SSID, WIFI_PASSWORD);
    Serial.print("Connexion au WiFi");
    while (WiFi.status() != WL_CONNECTED) {
        delay(500);
        Serial.print(".");
    }
    Serial.println("\nWiFi connecté ! IP : " + WiFi.localIP().toString());

    // Login API
    if (!login.LoginDispenser()) {
        Serial.println("Login échoué, continuer quand même");
    }

    // Connexion MQTT
    updater.connectMQTT();

    Serial.println("--- Test du capteur lancé ---");
}

void loop() {
    // Vérifie la connexion MQTT
    if (!mqtt.connected()) updater.connectMQTT();
    mqtt.loop();

    // Lecture du capteur
    bool isEmptyCurrent = !monCapteur.estDetecte(); // true si vide, false si liquide

    // Affichage console
    if (isEmptyCurrent) {
        Serial.println("ÉTAT : SEC...");
    } else {
        Serial.println("ÉTAT : LIQUIDE DÉTECTÉ !");
    }

    // Envoie l'update uniquement si l'état a changé
    if (isEmptyCurrent != lastIsEmpty) {
        updater.sendStatus(isEmptyCurrent);
        lastIsEmpty = isEmptyCurrent; // met à jour l'état précédent
    }

    delay(500);
}
